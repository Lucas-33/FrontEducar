"""Dependencias de autenticación y autorización para la API.

Centraliza la validación del token JWT y el control de acceso por rol,
para que todos los endpoints exijan sesión válida (y rol cuando corresponda).
"""

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from sqlalchemy.orm import Session

from backend.app.core.database import get_db
from backend.app.models.seguridad import Usuario
from backend.app.services.seguridad import seguridad_service

bearer_scheme = HTTPBearer(auto_error=False)


def get_current_user(
    credentials: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
    db: Session = Depends(get_db),
) -> Usuario:
    """Devuelve el usuario dueño del token Bearer o lanza 401/403."""
    if not credentials or not credentials.credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Falta el token de autenticación",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return seguridad_service.get_usuario_from_token(db, credentials.credentials)


def require_roles(*roles: str):
    """Genera una dependencia que exige que el usuario tenga uno de los roles dados."""

    def verificar_rol(
        current_user: Usuario = Depends(get_current_user),
        db: Session = Depends(get_db),
    ) -> Usuario:
        rol = seguridad_service.get_rol_nombre(db, current_user)
        if rol not in roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="No tiene permisos para realizar esta acción",
            )
        return current_user

    return verificar_rol

def _contexto_acceso(db: Session, current_user: Usuario):
    """Devuelve (rol, legajos_permitidos) según el usuario autenticado.

    legajos_permitidos = None significa "sin restricción" (ADMIN/DOCENTE).
    """
    rol = seguridad_service.get_rol_nombre(db, current_user)
    if rol == "STUDENT":
        if not current_user.legajo_alumno:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="El usuario no tiene un alumno vinculado",
            )
        return rol, [current_user.legajo_alumno]
    if rol == "PARENT":
        return rol, evaluacion_service.legajos_de_padre(db, current_user.id_padre)
    return rol, None

ENCABEZADO_FILL = PatternFill(
    start_color="0D6E6E", end_color="0D6E6E", fill_type="solid"
)
ENCABEZADO_FONT = Font(color="FFFFFF", bold=True)


def _autorizar_admin(db: Session, token: str | None) -> Usuario:
    if not token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Falta el token de autenticación",
        )
    usuario = seguridad_service.get_usuario_from_token(db, token)
    if seguridad_service.get_rol_nombre(db, usuario) != "ADMIN":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Solo los administradores pueden exportar reportes",
        )
    return usuario


def _agregar_hoja(wb: Workbook, titulo: str, encabezados: list[str], filas: list[list]):
    ws = wb.create_sheet(title=titulo[:31])
    ws.append(encabezados)
    for celda in ws[1]:
        celda.fill = ENCABEZADO_FILL
        celda.font = ENCABEZADO_FONT
        celda.alignment = Alignment(horizontal="center")
    for fila in filas:
        ws.append(fila)
    for indice, _ in enumerate(encabezados, start=1):
        letra = get_column_letter(indice)
        ancho = max(
            (
                len(str(ws.cell(row=fila, column=indice).value or ""))
                for fila in range(1, ws.max_row + 1)
            ),
            default=10,
        )
        ws.column_dimensions[letra].width = min(ancho + 4, 48)
    ws.freeze_panes = "A2"


def _respuesta_xlsx(wb: Workbook, nombre_archivo: str) -> StreamingResponse:
    buffer = BytesIO()
    wb.save(buffer)
    buffer.seek(0)
    return StreamingResponse(
        buffer,
        media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={"Content-Disposition": f'attachment; filename="{nombre_archivo}"'},
    )


def _nombre_curso(db: Session, id_curso: int) -> str:
    curso = curso_repository.get(db, id_curso)
    return curso.nombre if curso else f"Curso #{id_curso}"


def _nombre_nivel_de_curso(db: Session, id_curso: int) -> str:
    curso = curso_repository.get(db, id_curso)
    if not curso:
        return "-"
    nivel = nivel_repository.get(db, curso.id_nivel)
    return nivel.nombre if nivel else "-"

@router.get(

import os

from dotenv import load_dotenv
from sqlalchemy import create_engine
from sqlalchemy.orm import declarative_base, sessionmaker

load_dotenv()

DATABASE_URL = os.getenv("DATABASE_URL")

if not DATABASE_URL:
    raise ValueError("No se encontró DATABASE_URL en el archivo .env")

engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

Base = declarative_base()


# Dependencia para inyectar en los endpoints de FastAPI
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

from typing import Any, Generic, TypeVar

from pydantic import BaseModel
from sqlalchemy import select
from sqlalchemy.orm import Session

from backend.app.core.database import Base

ModelType = TypeVar("ModelType", bound=Base)
CreateSchemaType = TypeVar("CreateSchemaType", bound=BaseModel)
UpdateSchemaType = TypeVar("UpdateSchemaType", bound=BaseModel)


class CRUDBase(Generic[ModelType, CreateSchemaType, UpdateSchemaType]):
    def __init__(self, model: type[ModelType]):
        self.model = model

    def get(self, db: Session, id_val: Any) -> ModelType | None:
        return db.get(self.model, id_val)

    def get_multi(
        self, db: Session, skip: int = 0, limit: int = 100
    ) -> list[ModelType]:
        stmt = select(self.model).offset(skip).limit(limit)
        return list(db.scalars(stmt).all())

    def create(self, db: Session, obj_in: CreateSchemaType) -> ModelType:
        obj_data = obj_in.model_dump()
        db_obj = self.model(**obj_data)
        db.add(db_obj)
        db.commit()
        db.refresh(db_obj)
        return db_obj

    def update(
        self, db: Session, db_obj: ModelType, obj_in: UpdateSchemaType | dict[str, Any]
    ) -> ModelType:
        obj_data = (
            obj_in
            if isinstance(obj_in, dict)
            else obj_in.model_dump(exclude_unset=True)
        )
        for field, value in obj_data.items():
            setattr(db_obj, field, value)
        db.commit()
        db.refresh(db_obj)
        return db_obj

    def remove(self, db: Session, id_val: Any) -> ModelType | None:
        obj = db.get(self.model, id_val)
        if obj:
            db.delete(obj)
            db.commit()
        return obj

    def create_user_with_hash(
        self, db: Session, obj_in: UsuarioCreate, password_hash: str
    ) -> Usuario:
        user_data = obj_in.model_dump(exclude={"password"})
        user_data["password_hash"] = password_hash
        db_user = Usuario(**user_data)
        db.add(db_user)
        db.commit()
        db.refresh(db_user)
        return db_user

    raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail=mensaje)

    def update_asignacion(
        self,
        db: Session,
        id_materia: int,
        id_curso: int,
        asignacion_in: MateriaCursoProfesorUpdate,
    ):
        existente = materia_curso_profesor_repository.get_by_materia_y_curso(
            db, id_materia, id_curso
        )
        if not existente:
            raise HTTPException(
                status_code=404,
                detail=f"No existe la asignación materia #{id_materia} / curso #{id_curso}",
            )
        datos = asignacion_in.model_dump(exclude_unset=True)
        nuevo_id_materia = datos.get("id_materia", id_materia)
        nuevo_id_curso = datos.get("id_curso", id_curso)
        nuevo_legajo = datos.get("legajo_profesor", existente.legajo_profesor)

        if not materia_repository.get(db, nuevo_id_materia):
            raise HTTPException(
                status_code=404, detail="La materia especificada no existe"
            )
        if not curso_repository.get(db, nuevo_id_curso):
            raise HTTPException(
                status_code=404, detail="El curso especificado no existe"
            )
        if not profesor_repository.get_by_legajo(db, nuevo_legajo):
            raise HTTPException(
                status_code=404, detail="El profesor especificado no existe"
            )

        if (nuevo_id_materia, nuevo_id_curso) != (id_materia, id_curso):
            # La clave primaria no se puede modificar en sitio: se elimina y se recrea
            duplicado = materia_curso_profesor_repository.get_by_materia_y_curso(
                db, nuevo_id_materia, nuevo_id_curso
            )
            if duplicado:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="La materia y el curso destino ya tienen una asignación existente",
                )
            materia_curso_profesor_repository.remove(db, (id_materia, id_curso))
            return materia_curso_profesor_repository.create(
                db,
                MateriaCursoProfesorCreate(
                    id_materia=nuevo_id_materia,
                    id_curso=nuevo_id_curso,
                    legajo_profesor=nuevo_legajo,
                ),
            )

        return materia_curso_profesor_repository.update(
            db, existente, {"legajo_profesor": nuevo_legajo}
        )

    def delete_asignacion(self, db: Session, id_materia: int, id_curso: int):
        existente = materia_curso_profesor_repository.get_by_materia_y_curso(

            nivel = nivel_repository.get(db, curso.id_nivel) if curso else None
            profesor = profesor_repository.get_by_legajo(db, asignacion.legajo_profesor)

from datetime import date

from fastapi import HTTPException
from sqlalchemy.orm import Session

from backend.app.repositories import (

class EvaluacionService:
    # ================= CALIFICACIONES =================
    def create_calificacion(
        self,
        db: Session,
        calificacion_in: CalificacionCreate,
        legajo_profesor: int | None = None,
    ):
        if not alumno_repository.get_by_legajo(db, calificacion_in.legajo_alumno):
            raise HTTPException(status_code=404, detail="El alumno no existe")
        if not materia_repository.get(db, calificacion_in.id_materia):
            raise HTTPException(status_code=404, detail="La materia no existe")
        if not curso_repository.get(db, calificacion_in.id_curso):
            raise HTTPException(status_code=404, detail="El curso no existe")

        datos = calificacion_in.model_dump()
        datos["legajo_profesor"] = legajo_profesor
        if datos.get("fecha") is None:
            datos["fecha"] = date.today()

        from backend.app.models.evaluacion import Calificacion

        registro = Calificacion(**datos)
        db.add(registro)
        db.commit()
        db.refresh(registro)
        return registro

    def get_calificacion(self, db: Session, id_calificacion: int):
        registro = calificacion_repository.get(db, id_calificacion)
        if not registro:
            raise HTTPException(
                status_code=404, detail=f"La calificación #{id_calificacion} no existe"
            )
        return registro

    def update_calificacion(
        self, db: Session, id_calificacion: int, calificacion_in: CalificacionUpdate
    ):
        registro = self.get_calificacion(db, id_calificacion)
        return calificacion_repository.update(db, registro, calificacion_in)

    def delete_calificacion(self, db: Session, id_calificacion: int):
        registro = self.get_calificacion(db, id_calificacion)
        calificacion_repository.remove(db, id_calificacion)
        return registro

    def _nombre_alumno(self, db: Session, legajo: int) -> str:
        alumno = alumno_repository.get_by_legajo(db, legajo)
        if not alumno:
            return f"Legajo #{legajo}"
        return f"{alumno.apellido}, {alumno.nombre}"

    def get_calificaciones_detalle(
        self,
        db: Session,
        legajo_alumno: int | None = None,
        legajos_permitidos: list[int] | None = None,
        id_materia: int | None = None,
        id_curso: int | None = None,
        periodo: str | None = None,
    ) -> list[CalificacionDetalleResponse]:
        if legajo_alumno is not None:
            registros = calificacion_repository.get_by_alumno(db, legajo_alumno)
        elif id_materia is not None and id_curso is not None:
            registros = calificacion_repository.get_by_materia_curso(
                db, id_materia, id_curso
            )
        elif id_curso is not None:
            registros = calificacion_repository.get_by_curso(db, id_curso)
        else:
            registros = calificacion_repository.get_multi(db, limit=1000)

        if legajos_permitidos is not None:
            registros = [r for r in registros if r.legajo_alumno in legajos_permitidos]
        if id_materia is not None:
            registros = [r for r in registros if r.id_materia == id_materia]
        if id_curso is not None:
            registros = [r for r in registros if r.id_curso == id_curso]
        if periodo:
            registros = [r for r in registros if r.periodo == periodo]

        resultado: list[CalificacionDetalleResponse] = []
        cache_materias = {}
        cache_cursos = {}
        for r in registros:
            if r.id_materia not in cache_materias:
                materia = materia_repository.get(db, r.id_materia)
                cache_materias[r.id_materia] = (
                    materia.nombre if materia else f"Materia #{r.id_materia}"
                )
            if r.id_curso not in cache_cursos:
                curso = curso_repository.get(db, r.id_curso)
                cache_cursos[r.id_curso] = (
                    curso.nombre if curso else f"Curso #{r.id_curso}"
                )
            resultado.append(
                CalificacionDetalleResponse(
                    id_calificacion=r.id_calificacion,
                    legajo_alumno=r.legajo_alumno,
                    alumno=self._nombre_alumno(db, r.legajo_alumno),
                    id_materia=r.id_materia,
                    materia=cache_materias[r.id_materia],
                    id_curso=r.id_curso,
                    curso=cache_cursos[r.id_curso],
                    periodo=r.periodo,
                    tipo=r.tipo,
                    nota=float(r.nota),
                    fecha=r.fecha,
                    observacion=r.observacion,
                )
            )
        return resultado

    # ================= ASISTENCIA =================
    def registrar_asistencia_lote(
        self,
        db: Session,
        lote_in: AsistenciaLoteCreate,
        legajo_profesor: int | None = None,
    ) -> list[AsistenciaDetalleResponse]:
        if not curso_repository.get(db, lote_in.id_curso):
            raise HTTPException(status_code=404, detail="El curso no existe")

        fecha = lote_in.fecha or date.today()
        for item in lote_in.registros:
            if not alumno_repository.get_by_legajo(db, item.legajo_alumno):
                raise HTTPException(
                    status_code=404,
                    detail=f"El alumno con legajo #{item.legajo_alumno} no existe",
                )
            existente = asistencia_repository.get_registro(
                db, item.legajo_alumno, fecha
            )
            if existente:
                existente.estado = item.estado
                existente.observacion = item.observacion
                existente.id_curso = lote_in.id_curso
                existente.legajo_profesor = legajo_profesor
            else:
                db.add(
                    asistencia_repository.model(
                        legajo_alumno=item.legajo_alumno,
                        id_curso=lote_in.id_curso,
                        fecha=fecha,
                        estado=item.estado,
                        observacion=item.observacion,
                        legajo_profesor=legajo_profesor,
                    )
                )
        db.commit()
        return self.get_asistencia_detalle(db, id_curso=lote_in.id_curso, fecha=fecha)

    def get_asistencia_detalle(
        self,
        db: Session,
        id_curso: int | None = None,
        fecha: date | None = None,
        legajo_alumno: int | None = None,
        legajos_permitidos: list[int] | None = None,
    ) -> list[AsistenciaDetalleResponse]:
        if legajo_alumno is not None:
            registros = asistencia_repository.get_by_alumno(db, legajo_alumno)
        else:
            registros = asistencia_repository.get_multi(db, limit=2000)

        if id_curso is not None:
            registros = [r for r in registros if r.id_curso == id_curso]
        if fecha is not None:
            registros = [r for r in registros if r.fecha == fecha]
        if legajos_permitidos is not None:
            registros = [r for r in registros if r.legajo_alumno in legajos_permitidos]

        cache_cursos = {}
        resultado: list[AsistenciaDetalleResponse] = []
        for r in registros:
            if r.id_curso not in cache_cursos:
                curso = curso_repository.get(db, r.id_curso)
                cache_cursos[r.id_curso] = (
                    curso.nombre if curso else f"Curso #{r.id_curso}"
                )
            resultado.append(
                AsistenciaDetalleResponse(
                    id_asistencia=r.id_asistencia,
                    legajo_alumno=r.legajo_alumno,
                    alumno=self._nombre_alumno(db, r.legajo_alumno),
                    id_curso=r.id_curso,
                    curso=cache_cursos[r.id_curso],
                    fecha=r.fecha,
                    estado=r.estado,
                    observacion=r.observacion,
                )
            )
        resultado.sort(key=lambda x: x.fecha, reverse=True)
        return resultado

    def delete_asistencia(self, db: Session, id_asistencia: int):

    def resumen_asistencia(
        self, db: Session, legajo_alumno: int
    ) -> ResumenAsistenciaResponse:
        if not alumno_repository.get_by_legajo(db, legajo_alumno):
            raise HTTPException(status_code=404, detail="El alumno no existe")
        registros = asistencia_repository.get_by_alumno(db, legajo_alumno, limit=2000)
        total = len(registros)
        presentes = sum(1 for r in registros if r.estado == "PRESENTE")
        ausentes = sum(1 for r in registros if r.estado == "AUSENTE")
        tardes = sum(1 for r in registros if r.estado == "TARDE")
        justificados = sum(1 for r in registros if r.estado == "JUSTIFICADO")
        porcentaje = (
            round(((presentes + tardes + justificados) / total) * 100, 1)
            if total
            else 0.0
        )
        return ResumenAsistenciaResponse(
            legajo_alumno=legajo_alumno,
            total=total,
            presentes=presentes,
            ausentes=ausentes,
            tardes=tardes,
            justificados=justificados,
            porcentaje_presentismo=porcentaje,
        )

    # Útil para endpoints: hijos del tutor autenticado
    def legajos_de_padre(self, db: Session, id_padre: int | None) -> list[int]:
        if not id_padre:
            return []
        return [
            rel.legajo_alumno
            for rel in padre_alumno_repository.get_alumnos_por_padre(db, id_padre)
        ]

import os
from datetime import datetime, timedelta, timezone

import bcrypt
import jwt
from fastapi import HTTPException, status
from sqlalchemy.orm import Session

from backend.app.models.seguridad import Usuario
from backend.app.repositories import (
    alumno_repository,
    padre_repository,

    usuario_repository,

from backend.app.schemas.seguridad import (

    UsuarioAdminResponse,

    UsuarioUpdate,

SECRET_KEY = os.getenv("SECRET_KEY")
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = int(
    os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "1440")
)  # 24 horas


class SeguridadService:
    @staticmethod
    def verify_password(plain_password: str, hashed_password: str) -> bool:
        """Verifica una contraseña plana contra el hash bcrypt almacenado."""
        if not plain_password or not hashed_password:
            return False
        try:
            password_bytes = plain_password.encode("utf-8")[:72]
            hashed_bytes = (
                hashed_password.encode("utf-8")
                if isinstance(hashed_password, str)
                else hashed_password
            )
            return bcrypt.checkpw(password_bytes, hashed_bytes)
        except Exception:
            return False

    @staticmethod
    def get_password_hash(password: str) -> str:
        """Genera un hash bcrypt seguro en formato string ($2b$)."""
        password_bytes = password.encode("utf-8")[:72]
        salt = bcrypt.gensalt()
        return bcrypt.hashpw(password_bytes, salt).decode("utf-8")

    @staticmethod
    def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
        to_encode = data.copy()
        expire = datetime.now(timezone.utc) + (
            expires_delta or timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
        )
        to_encode.update({"exp": expire})
        return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

    def authenticate_user(self, db: Session, username: str, password: str):
        user = usuario_repository.get_by_username(db, username=username)
        if not user or not self.verify_password(password, user.password_hash):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Credenciales incorrectas",
                headers={"WWW-Authenticate": "Bearer"},
            )
        if not user.activo:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="El usuario se encuentra inactivo",
            )
        return user

    def login(self, db: Session, username: str, password: str) -> Token:
        user = self.authenticate_user(db, username, password)
        rol = rol_repository.get(db, user.id_rol)
        rol_nombre = rol.nombre if rol else "USER"

        token_data = {
            "sub": user.username,
            "id_usuario": user.id_usuario,
            "rol": rol_nombre,
        }
        access_token = self.create_access_token(data=token_data)
        return Token(
            access_token=access_token,
            token_type="bearer",
            rol=rol_nombre,
            username=user.username,
        )

    def get_usuario_from_token(self, db: Session, token: str) -> Usuario:
        """Decodifica el JWT y devuelve la entidad Usuario. Lanza 401/403 si no es válido."""
        try:
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        except jwt.ExpiredSignatureError:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="El token expiró. Inicie sesión nuevamente.",
                headers={"WWW-Authenticate": "Bearer"},
            )
        except jwt.InvalidTokenError:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Token inválido",
                headers={"WWW-Authenticate": "Bearer"},
            )

        username = payload.get("sub")
        user = usuario_repository.get_by_username(db, username) if username else None
        if not user:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="El usuario del token no existe",
                headers={"WWW-Authenticate": "Bearer"},
            )
        if not user.activo:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="El usuario se encuentra inactivo",
            )
        return user

    def get_rol_nombre(self, db: Session, user: Usuario) -> str:
        """Devuelve el nombre del rol (ADMIN, TEACHER, PARENT, STUDENT) del usuario."""
        rol = rol_repository.get(db, user.id_rol)
        return rol.nombre if rol else ""

    def get_perfil_from_token(self, db: Session, token: str) -> UsuarioAdminResponse:
        """Decodifica el JWT y devuelve el perfil del usuario autenticado."""
        user = self.get_usuario_from_token(db, token)
        return self._to_admin_response(db, user)

    def change_password(
        self, db: Session, user: Usuario, password_actual: str, password_nueva: str
    ) -> dict:
        """Cambia la contraseña del propio usuario autenticado."""
        if not self.verify_password(password_actual, user.password_hash):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="La contraseña actual no es correcta",
            )
        if len(password_nueva or "") < 6:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="La nueva contraseña debe tener al menos 6 caracteres",
            )
        user.password_hash = self.get_password_hash(password_nueva)
        db.commit()
        return {"detail": "Contraseña actualizada correctamente"}

    def registrar_usuario(self, db: Session, user_in: UsuarioCreate):
        if usuario_repository.get_by_username(db, user_in.username):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"El nombre de usuario '{user_in.username}' ya está registrado",
            )
        if not rol_repository.get(db, user_in.id_rol):
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El rol con ID {user_in.id_rol} no existe",
            )
        hashed_pwd = self.get_password_hash(user_in.password)
        return usuario_repository.create_user_with_hash(db, user_in, hashed_pwd)

    # ================= ADMINISTRACIÓN DE USUARIOS =================
    def get_roles(self, db: Session):
        return rol_repository.get_multi(db)

    def _to_admin_response(self, db: Session, user) -> UsuarioAdminResponse:
        rol = rol_repository.get(db, user.id_rol)
        rol_nombre = rol.nombre if rol else ""

        if user.legajo_alumno:
            alumno = alumno_repository.get_by_legajo(db, user.legajo_alumno)
            vinculacion = (
                f"Alumno: {alumno.apellido}, {alumno.nombre} (#{alumno.legajo})"
                if alumno
                else f"Alumno #{user.legajo_alumno}"
            )
        elif user.legajo_profesor:
            profesor = profesor_repository.get_by_legajo(db, user.legajo_profesor)
            vinculacion = (
                f"Docente: {profesor.apellido}, {profesor.nombre} (#{profesor.legajo})"
                if profesor
                else f"Docente #{user.legajo_profesor}"
            )
        elif user.id_padre:
            padre = padre_repository.get(db, user.id_padre)
            vinculacion = (
                f"Tutor: {padre.apellido}, {padre.nombre}"
                if padre
                else f"Tutor #{user.id_padre}"
            )
        elif rol_nombre == "ADMIN":
            vinculacion = "Acceso Total"
        else:
            vinculacion = "Sin vínculo"

        return UsuarioAdminResponse(
            id_usuario=user.id_usuario,
            username=user.username,
            id_rol=user.id_rol,
            legajo_alumno=user.legajo_alumno,
            legajo_profesor=user.legajo_profesor,
            id_padre=user.id_padre,
            activo=user.activo,
            creado_en=user.creado_en,
            rol=rol_nombre,
            vinculacion=vinculacion,
        )

    def get_usuarios(self, db: Session) -> list[UsuarioAdminResponse]:
        usuarios = usuario_repository.get_multi(db, limit=200)
        return [self._to_admin_response(db, u) for u in usuarios]

    def update_usuario(
        self, db: Session, id_usuario: int, user_in: UsuarioUpdate
    ) -> UsuarioAdminResponse:
        user = usuario_repository.get(db, id_usuario)
        if not user:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El usuario con ID {id_usuario} no existe",
            )

        data = user_in.model_dump(exclude_unset=True)
        password = data.pop("password", None)
        if password:
            user.password_hash = self.get_password_hash(password)

        nuevo_username = data.get("username")
        if nuevo_username and nuevo_username != user.username:
            if usuario_repository.get_by_username(db, nuevo_username):
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=f"El nombre de usuario '{nuevo_username}' ya está registrado",
                )

        nuevo_rol = data.get("id_rol")
        if nuevo_rol is not None and not rol_repository.get(db, nuevo_rol):
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El rol con ID {nuevo_rol} no existe",
            )

        for field, value in data.items():
            if value is not None:
                setattr(user, field, value)

        db.commit()
        db.refresh(user)
        return self._to_admin_response(db, user)

    def get_deportes(self, db: Session):
        return deporte_repository.get_multi(db)

    def update_deporte(self, db: Session, id_deporte: int, deporte_in: DeporteUpdate):
        deporte = deporte_repository.get(db, id_deporte)
        if not deporte:
            raise HTTPException(
                status_code=404, detail=f"El deporte con ID {id_deporte} no existe"
            )
        return deporte_repository.update(db, deporte, deporte_in)

    def delete_deporte(self, db: Session, id_deporte: int):
        deporte = deporte_repository.get(db, id_deporte)
        if not deporte:
            raise HTTPException(
                status_code=404, detail=f"El deporte con ID {id_deporte} no existe"
            )
        try:
            deporte_repository.remove(db, id_deporte)
        except IntegrityError:
            db.rollback()
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="No se puede eliminar el deporte porque tiene grupos o inscriptos asociados",
            )
        return deporte

    def create_grupo_deporte(self, db: Session, grupo_in: GrupoDeporteCreate):
        if not deporte_repository.get(db, grupo_in.id_deporte):
            raise HTTPException(status_code=404, detail="Deporte no encontrado")
        if not nivel_repository.get(db, grupo_in.id_nivel):
            raise HTTPException(status_code=404, detail="Nivel educativo no encontrado")
        if not horario_repository.get(db, grupo_in.id_horario):
            raise HTTPException(status_code=404, detail="Horario no encontrado")
        return grupo_deporte_repository.create(db, grupo_in)

    def update_grupo_deporte(
        self, db: Session, id_grupo: int, grupo_in: GrupoDeporteUpdate
    ):
        grupo = grupo_deporte_repository.get(db, id_grupo)
        if not grupo:
            raise HTTPException(
                status_code=404, detail=f"El grupo con ID {id_grupo} no existe"
            )
        datos = grupo_in.model_dump(exclude_unset=True)
        if datos.get("id_deporte") is not None and not deporte_repository.get(
            db, datos["id_deporte"]
        ):
            raise HTTPException(status_code=404, detail="Deporte no encontrado")
        if datos.get("id_nivel") is not None and not nivel_repository.get(
            db, datos["id_nivel"]
        ):
            raise HTTPException(status_code=404, detail="Nivel educativo no encontrado")
        if datos.get("id_horario") is not None and not horario_repository.get(
            db, datos["id_horario"]
        ):
            raise HTTPException(status_code=404, detail="Horario no encontrado")
        return grupo_deporte_repository.update(db, grupo, grupo_in)

    def delete_grupo_deporte(self, db: Session, id_grupo: int):
        grupo = grupo_deporte_repository.get(db, id_grupo)
        if not grupo:
            raise HTTPException(
                status_code=404, detail=f"El grupo con ID {id_grupo} no existe"
            )
        try:
            grupo_deporte_repository.remove(db, id_grupo)
        except IntegrityError:
            db.rollback()
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="No se puede eliminar el grupo porque tiene alumnos inscriptos",
            )
        return grupo

    def inscribir_deporte(self, db: Session, inscripcion_in: InscripcionDeporteCreate):
        if not alumno_repository.get_by_legajo(db, inscripcion_in.legajo_alumno):
            raise HTTPException(status_code=404, detail="El alumno no existe")

        nuevo_grupo = grupo_deporte_repository.get(db, inscripcion_in.id_grupo)
        if not nuevo_grupo:
            raise HTTPException(status_code=404, detail="El grupo deportivo no existe")

        if inscripcion_deporte_repository.get_inscripcion(
            db, inscripcion_in.legajo_alumno, inscripcion_in.id_grupo
        ):
            raise HTTPException(
                status_code=400, detail="El alumno ya está inscripto en este grupo"
            )

        # 1. Validar límite máximo de 2 deportes (RF03)
        inscripciones_actuales = inscripcion_deporte_repository.get_by_alumno(
            db, inscripcion_in.legajo_alumno
        )
        if len(inscripciones_actuales) >= 2:
            raise HTTPException(
                status_code=400,
                detail="El alumno ya alcanzó el límite máximo de dos deportes simultáneos",
            )

        # 2. Controlar conflictos de horario (RF07)
        for inscripcion in inscripciones_actuales:
            grupo_actual = grupo_deporte_repository.get(db, inscripcion.id_grupo)
            if grupo_actual and grupo_actual.id_horario == nuevo_grupo.id_horario:
                raise HTTPException(
                    status_code=409,
                    detail="Existe un conflicto de horarios con otro deporte en el que el alumno ya está inscripto",
                )

        return inscripcion_deporte_repository.create(db, inscripcion_in)

    # --- TRANSPORTE ---
    def create_recorrido(self, db: Session, recorrido_in: RecorridoTransporteCreate):

    def update_recorrido(
        self, db: Session, id_recorrido: int, recorrido_in: RecorridoTransporteUpdate
    ):
        recorrido = recorrido_transporte_repository.get(db, id_recorrido)
        if not recorrido:
            raise HTTPException(
                status_code=404, detail=f"El recorrido con ID {id_recorrido} no existe"
            )
        return recorrido_transporte_repository.update(db, recorrido, recorrido_in)

    def delete_recorrido(self, db: Session, id_recorrido: int):
        recorrido = recorrido_transporte_repository.get(db, id_recorrido)
        if not recorrido:
            raise HTTPException(
                status_code=404, detail=f"El recorrido con ID {id_recorrido} no existe"
            )
        try:
            recorrido_transporte_repository.remove(db, id_recorrido)
        except IntegrityError:
            db.rollback()
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="No se puede eliminar el recorrido porque tiene pasajeros asociados",
            )
        return recorrido

    def inscribir_transporte(
        self, db: Session, inscripcion_in: InscripcionTransporteCreate
    ):
        if not alumno_repository.get_by_legajo(db, inscripcion_in.legajo_alumno):
            raise HTTPException(status_code=404, detail="El alumno no existe")
        if not recorrido_transporte_repository.get(db, inscripcion_in.id_recorrido):
            raise HTTPException(status_code=404, detail="El recorrido no existe")

        existente = inscripcion_transporte_repository.get_by_alumno(
            db, inscripcion_in.legajo_alumno
        )
        if existente and existente.estado == "ACTIVO":
            raise HTTPException(
                status_code=400,
                detail="El alumno ya cuenta con inscripción activa en transporte",
            )
        return inscripcion_transporte_repository.create(db, inscripcion_in)

    def baja_transporte(self, db: Session, legajo_alumno: int):
        res = inscripcion_transporte_repository.soft_delete(db, legajo_alumno)
        if not res:
            raise HTTPException(
                status_code=404, detail="Inscripción de transporte no encontrada"
            )
        return res

    # --- COMEDOR ---
    def inscribir_comedor(self, db: Session, inscripcion_in: InscripcionComedorCreate):
        if not alumno_repository.get_by_legajo(db, inscripcion_in.legajo_alumno):

    client,
    cabeceras_profesor,
    cabeceras_alumno,
    cabeceras_padre,
    cabeceras_admin,
):

    lote = client.post(
        "/api/v1/academico/asistencia/lote",

            "id_curso": 1,

            "registros": [
                {"legajo_alumno": LEGAJO_ALUMNO_1, "estado": "PRESENTE"},

            ],

        headers=cabeceras_profesor,
    )

    assert len(lote.json()) == 2

        "/api/v1/academico/asistencia/lote",

            "id_curso": 1,
            "fecha": fecha,

                    "estado": "TARDE",
                    "observacion": "Llegó tarde",
                },
            ],
        },
        headers=cabeceras_profesor,
    )
    assert relote.status_code == 200
    assert len(relote.json()) == 2
    registro_1 = next(r for r in relote.json() if r["legajo_alumno"] == LEGAJO_ALUMNO_1)
    assert registro_1["estado"] == "TARDE"
    id_registro_1 = registro_1["id_asistencia"]

    # El alumno solo ve su propia asistencia
    propias = client.get("/api/v1/academico/asistencia", headers=cabeceras_alumno)
    assert propias.status_code == 200
    assert all(r["legajo_alumno"] == LEGAJO_ALUMNO_1 for r in propias.json())

    # El tutor ve la de su hijo, pero no la de otro alumno
    assert (
        client.get(
            f"/api/v1/academico/asistencia?legajo_alumno={LEGAJO_ALUMNO_1}",
            headers=cabeceras_padre,

            headers=cabeceras_padre,

        == 403
    )

        f"/api/v1/academico/asistencia/resumen/{LEGAJO_ALUMNO_1}",
        headers=cabeceras_alumno,
    )

            f"/api/v1/academico/asistencia/resumen/{LEGAJO_ALUMNO_2}",
            headers=cabeceras_padre,
        ).status_code
        == 403
    )

    # Eliminar el registro como ADMIN lo quita del listado
    eliminar = client.delete(
        f"/api/v1/academico/asistencia/{id_registro_1}",
        headers=cabeceras_admin,
    )
    assert eliminar.status_code == 200
    restantes = client.get(
        f"/api/v1/academico/asistencia?id_curso=1&fecha={fecha}",
        headers=cabeceras_admin,
    ).json()
    assert all(r["id_asistencia"] != id_registro_1 for r in restantes)

def test_limite_deportes_y_conflictos_horario(client, cabeceras_admin):
    # 1. Crear turnos/horarios de prueba
    h1 = client.post(
        "/api/v1/academico/horarios",
        json={"descripcion": "Turno Mañana"},
        headers=cabeceras_admin,
    ).json()["id_horario"]
    h2 = client.post(
        "/api/v1/academico/horarios",
        json={"descripcion": "Turno Tarde"},
        headers=cabeceras_admin,
    ).json()["id_horario"]

    # 2. Crear deportes
    d_futbol = client.post(
        "/api/v1/servicios/deportes", json={"nombre": "Fútbol"}, headers=cabeceras_admin
    ).json()["id_deporte"]
    d_voley = client.post(
        "/api/v1/servicios/deportes", json={"nombre": "Vóley"}, headers=cabeceras_admin
    ).json()["id_deporte"]
    d_basquet = client.post(
        "/api/v1/servicios/deportes",
        json={"nombre": "Básquet"},
        headers=cabeceras_admin,
    ).json()["id_deporte"]

    # 3. Crear grupos deportivos (id_nivel=1 ya existe gracias a conftest.py)
    # Grupo 1: Fútbol a la Mañana
    g_futbol_manana = client.post(
        "/api/v1/servicios/deportes/grupos",
        json={"id_deporte": d_futbol, "id_nivel": 1, "id_horario": h1},
        headers=cabeceras_admin,
    ).json()["id_grupo"]

    # Grupo 2: Vóley a la Tarde
    g_voley_tarde = client.post(
        "/api/v1/servicios/deportes/grupos",
        json={"id_deporte": d_voley, "id_nivel": 1, "id_horario": h2},
        headers=cabeceras_admin,
    ).json()["id_grupo"]

    # Grupo 3: Básquet a la Mañana (¡Mismo horario que Fútbol!)
    g_basquet_manana = client.post(
        "/api/v1/servicios/deportes/grupos",
        json={"id_deporte": d_basquet, "id_nivel": 1, "id_horario": h1},
        headers=cabeceras_admin,
    ).json()["id_grupo"]

    # Grupo 4: Básquet a la Tarde (Diferente horario)
    g_basquet_tarde = client.post(
        "/api/v1/servicios/deportes/grupos",
        json={"id_deporte": d_basquet, "id_nivel": 1, "id_horario": h2},
        headers=cabeceras_admin,
    ).json()["id_grupo"]

    # 4. Inscribir al primer deporte (Fútbol a la Mañana) -> DEBE FUNCIONAR (201)
    resp1 = client.post(
        "/api/v1/servicios/deportes/inscripciones",
        json={"legajo_alumno": LEGAJO_ALUMNO_1, "id_grupo": g_futbol_manana},
        headers=cabeceras_admin,
    )
    assert resp1.status_code == 201

    # 5. Inscribir a Básquet a la Mañana -> DEBE FALLAR POR CONFLICTO DE HORARIO (409)
    resp2 = client.post(
        "/api/v1/servicios/deportes/inscripciones",
        json={"legajo_alumno": LEGAJO_ALUMNO_1, "id_grupo": g_basquet_manana},
        headers=cabeceras_admin,
    )
    assert resp2.status_code == 409
    assert "conflicto de horarios" in resp2.json()["detail"].lower()

    # 6. Inscribir a Vóley a la Tarde -> DEBE FUNCIONAR (201)
    resp3 = client.post(
