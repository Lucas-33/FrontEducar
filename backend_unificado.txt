@asynccontextmanager
async def lifespan(app: FastAPI):
    iniciar_scheduler()  # Encendemos el motor de tareas automáticas
    yield  # La API queda corriendo...
    # (Acá iría código de apagado si hiciera falta)

    version="1.0.0",

"""Dependencias de autenticación y autorización para la API.

Centraliza la validación del token JWT y el control de acceso por rol,
para que todos los endpoints exijan sesión válida (y rol cuando corresponda).
"""

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from sqlalchemy.orm import Session

from backend.app.core.database import get_db
from backend.app.models.seguridad import Usuario
from backend.app.services.seguridad import seguridad_service

bearer_scheme = HTTPBearer(auto_error=False)


def get_current_user(
    credentials: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
    db: Session = Depends(get_db),
) -> Usuario:
    """Devuelve el usuario dueño del token Bearer o lanza 401/403."""
    if not credentials or not credentials.credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Falta el token de autenticación",
            headers={"WWW-Authenticate": "Bearer"},
        )
    return seguridad_service.get_usuario_from_token(db, credentials.credentials)


def require_roles(*roles: str):
    """Genera una dependencia que exige que el usuario tenga uno de los roles dados."""

    def verificar_rol(
        current_user: Usuario = Depends(get_current_user),
        db: Session = Depends(get_db),
    ) -> Usuario:
        rol = seguridad_service.get_rol_nombre(db, current_user)
        if rol not in roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="No tiene permisos para realizar esta acción",
            )
        return current_user

    return verificar_rol

def _contexto_acceso(db: Session, current_user: Usuario):
    """Devuelve (rol, legajos_permitidos) según el usuario autenticado.

    legajos_permitidos = None significa "sin restricción" (ADMIN/DOCENTE).

    rol = seguridad_service.get_rol_nombre(db, current_user)

@router.get(
    "/facturas/{legajo_alumno}",
    response_model=list[FacturaResponse],
    summary="Historial de deuda/facturas (Padres)",
)
def listar_facturas(
    legajo_alumno: int,
    current_user: Usuario = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """Devuelve las facturas de un alumno, discriminando los ítems de deuda."""
    # Nota: Acá deberíamos validar que el current_user (padre) sea el tutor de este legajo.

    facturas_db = (
        db.query(Factura)
        .filter(Factura.legajo_alumno == legajo_alumno)
        .order_by(Factura.id_factura.desc())
        .all()
    )

    resultado = []
    for f in facturas_db:
        items_db = (
            db.query(ItemFactura).filter(ItemFactura.id_factura == f.id_factura).all()
        )

        # Convertimos la BD al Schema de respuesta
        factura_dict = f.__dict__.copy()
        factura_dict["items"] = [
            ItemFacturaResponse.model_validate(i) for i in items_db
        ]
        resultado.append(FacturaResponse(**factura_dict))

    return resultado


@router.post(
    "/comprobantes",
    status_code=status.HTTP_201_CREATED,
    summary="Subir comprobante de pago (Padres)",
)
def subir_comprobante(
    comprobante_in: ComprobantePagoCreate,
    current_user: Usuario = Depends(require_roles("PARENT")),
    db: Session = Depends(get_db),
):
    """Vincula un ticket de transferencia bancaria a una factura pendiente."""
    factura = (
        db.query(Factura)
        .filter(Factura.id_factura == comprobante_in.id_factura)
        .first()
    )
    if not factura:
        raise HTTPException(status_code=404, detail="La factura no existe")

    nuevo_comprobante = ComprobantePago(
        id_factura=comprobante_in.id_factura,
        archivo_url=comprobante_in.archivo_url,
        estado="PENDIENTE",  # El administrador web luego lo aprobará
    )
    db.add(nuevo_comprobante)
    db.commit()
    return {"mensaje": "Comprobante enviado exitosamente a la administración"}

ENCABEZADO_FILL = PatternFill(
    start_color="0D6E6E", end_color="0D6E6E", fill_type="solid"
)
ENCABEZADO_FONT = Font(color="FFFFFF", bold=True)


def _autorizar_admin(db: Session, token: str | None) -> Usuario:
    if not token:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Falta el token de autenticación",
        )
    usuario = seguridad_service.get_usuario_from_token(db, token)
    if seguridad_service.get_rol_nombre(db, usuario) != "ADMIN":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Solo los administradores pueden exportar reportes",
        )
    return usuario


def _agregar_hoja(wb: Workbook, titulo: str, encabezados: list[str], filas: list[list]):
    ws = wb.create_sheet(title=titulo[:31])
    ws.append(encabezados)
    for celda in ws[1]:
        celda.fill = ENCABEZADO_FILL
        celda.font = ENCABEZADO_FONT
        celda.alignment = Alignment(horizontal="center")
    for fila in filas:
        ws.append(fila)
    for indice, _ in enumerate(encabezados, start=1):
        letra = get_column_letter(indice)
        ancho = max(
            (
                len(str(ws.cell(row=fila, column=indice).value or ""))
                for fila in range(1, ws.max_row + 1)
            ),
            default=10,
        )
        ws.column_dimensions[letra].width = min(ancho + 4, 48)
    ws.freeze_panes = "A2"


def _respuesta_xlsx(wb: Workbook, nombre_archivo: str) -> StreamingResponse:
    buffer = BytesIO()
    wb.save(buffer)
    buffer.seek(0)
    return StreamingResponse(
        buffer,
        media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={"Content-Disposition": f'attachment; filename="{nombre_archivo}"'},
    )


def _nombre_curso(db: Session, id_curso: int) -> str:
    curso = curso_repository.get(db, id_curso)
    return curso.nombre if curso else f"Curso #{id_curso}"


def _nombre_nivel_de_curso(db: Session, id_curso: int) -> str:
    curso = curso_repository.get(db, id_curso)
    if not curso:
        return "-"
    nivel = nivel_repository.get(db, curso.id_nivel)
    return nivel.nombre if nivel else "-"

    raise ValueError("No se encontró DATABASE_URL en el archivo .env")

engine = create_engine(DATABASE_URL)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

Base = declarative_base()


# Dependencia para inyectar en los endpoints de FastAPI

    db = SessionLocal()
    try:
        yield db

        db.close()

scheduler = AsyncIOScheduler()


def es_ultimo_dia_habil() -> bool:
    """Calcula matemáticamente si hoy es el último día hábil (lunes a viernes) del mes."""
    hoy = date.today()
    ultimo_dia_del_mes = calendar.monthrange(hoy.year, hoy.month)[1]
    fecha_fin = date(hoy.year, hoy.month, ultimo_dia_del_mes)

    # Si el último día cae Sábado (5), el último hábil es el viernes anterior (-1 día)
    if fecha_fin.weekday() == 5:
        fecha_fin = date(hoy.year, hoy.month, ultimo_dia_del_mes - 1)
    # Si el último día cae Domingo (6), el último hábil es el viernes anterior (-2 días)
    elif fecha_fin.weekday() == 6:
        fecha_fin = date(hoy.year, hoy.month, ultimo_dia_del_mes - 2)

    return hoy == fecha_fin


def tarea_cierre_mensual():
    """Se ejecuta y revisa si es el último día hábil. Genera facturas y las envía por mail."""
    if not es_ultimo_dia_habil():
        return
        
    print("⏰ [SCHEDULER] Ejecutando cierre mensual y envío de facturas...")
    db = SessionLocal() 
    try:
        hoy = date.today()
        # 1. Generar facturas en BD
        finanzas_service.generar_facturas_mes(db, mes=hoy.month, anio=hoy.year)
        
        # 2. Buscar las recién creadas y enviarlas
        facturas = db.query(Factura).filter_by(mes=hoy.month, anio=hoy.year).all()
        for fac in facturas:
            alumno = db.query(Alumno).filter_by(legajo=fac.legajo_alumno).first()
            if alumno and alumno.correo_electronico:
                items = db.query(ItemFactura).filter_by(id_factura=fac.id_factura).all()
                ruta_pdf = generar_pdf_factura(fac, alumno, items)
                
                cuerpo = f"Estimada familia de {alumno.nombre},\n\nAdjuntamos la liquidación de servicios educativos correspondiente al mes {fac.mes}/{fac.anio}.\n\nSaludos cordiales,\nAdministración Educar."
                enviar_correo_con_pdf(
                    destinatario=alumno.correo_electronico, 
                    asunto=f"Nueva Liquidación Mensual - {alumno.apellido}", 
                    cuerpo=cuerpo, 
                    ruta_pdf=ruta_pdf
                )
    except Exception as e:
        print(f"❌ [SCHEDULER] Error en cierre mensual: {e}")
    finally:
        db.close()


def tarea_aviso_morosos():
    """Se ejecuta los días 20 de cada mes. Envía PDF con recordatorio de deuda."""
    print("⏰ [SCHEDULER] Revisando morosos para enviar aviso de deuda...")
    db = SessionLocal()
    try:
        # Buscar todas las facturas impagas
        facturas_adeudadas = db.query(Factura).filter(Factura.estado != "PAGADA").all()
        for fac in facturas_adeudadas:
            alumno = db.query(Alumno).filter_by(legajo=fac.legajo_alumno).first()
            if alumno and alumno.correo_electronico:
                items = db.query(ItemFactura).filter_by(id_factura=fac.id_factura).all()
                ruta_pdf = generar_pdf_factura(fac, alumno, items)
                
                cuerpo = f"Aviso Importante: Saldo Pendiente\n\nEstimada familia de {alumno.nombre},\nLe recordamos que posee un saldo vencido de $ {fac.total:.2f} en su cuenta.\nLe adjuntamos el detalle. Por favor, regularice su situación a la brevedad para evitar recargos.\n\nSaludos."
                enviar_correo_con_pdf(
                    destinatario=alumno.correo_electronico, 
                    asunto="Aviso de Deuda - Educar para Transformar", 
                    cuerpo=cuerpo, 
                    ruta_pdf=ruta_pdf
                )
    except Exception as e:
        print(f"❌ [SCHEDULER] Error avisando a morosos: {e}")
    finally:
        db.close()


def iniciar_scheduler():
    """Registra los disparadores de tiempo y enciende el motor."""
    # Disparador 1: Corre TODOS LOS DÍAS a las 08:00 AM (la función por dentro chequea si es el último día hábil)
    scheduler.add_job(tarea_cierre_mensual, CronTrigger(hour=8, minute=0))

    # Disparador 2: Corre SOLAMENTE el día 20 de cada mes a las 08:00 AM
    scheduler.add_job(tarea_aviso_morosos, CronTrigger(day=20, hour=8, minute=0))

    scheduler.start()
    print("🚀 Planificador de tareas (APScheduler) iniciado en segundo plano.")

from sqlalchemy import select
from sqlalchemy.orm import Session

from backend.app.models.academico import (
    Curso,
    Materia,
    MateriaCursoProfesor,
    NivelEducativo,
)
from backend.app.repositories.base import CRUDBase
from backend.app.schemas.academico import (
    CursoBase,
    CursoCreate,
    MateriaBase,
    MateriaCreate,
    MateriaCursoProfesorBase,
    MateriaCursoProfesorCreate,
    NivelEducativoBase,
    NivelEducativoCreate,
)


class NivelEducativoRepository(
    CRUDBase[NivelEducativo, NivelEducativoCreate, NivelEducativoBase]
):
    def __init__(self):
        super().__init__(NivelEducativo)


class CursoRepository(CRUDBase[Curso, CursoCreate, CursoBase]):
    def __init__(self):
        super().__init__(Curso)

    def get_by_nivel(self, db: Session, id_nivel: int) -> list[Curso]:
        stmt = select(Curso).where(Curso.id_nivel == id_nivel)
        return list(db.scalars(stmt).all())


class MateriaRepository(CRUDBase[Materia, MateriaCreate, MateriaBase]):
    def __init__(self):
        super().__init__(Materia)


class MateriaCursoProfesorRepository(
    CRUDBase[MateriaCursoProfesor, MateriaCursoProfesorCreate, MateriaCursoProfesorBase]
):
    def __init__(self):
        super().__init__(MateriaCursoProfesor)

    def get_by_materia_y_curso(
        self, db: Session, id_materia: int, id_curso: int
    ) -> MateriaCursoProfesor | None:
        stmt = select(MateriaCursoProfesor).where(
            MateriaCursoProfesor.id_materia == id_materia,
            MateriaCursoProfesor.id_curso == id_curso,
        )
        return db.scalars(stmt).first()

    def get_by_curso(self, db: Session, id_curso: int) -> list[MateriaCursoProfesor]:
        stmt = select(MateriaCursoProfesor).where(

from typing import Any, Generic, TypeVar

from pydantic import BaseModel
from sqlalchemy import select
from sqlalchemy.orm import Session

from backend.app.core.database import Base

ModelType = TypeVar("ModelType", bound=Base)
CreateSchemaType = TypeVar("CreateSchemaType", bound=BaseModel)
UpdateSchemaType = TypeVar("UpdateSchemaType", bound=BaseModel)


class CRUDBase(Generic[ModelType, CreateSchemaType, UpdateSchemaType]):
    def __init__(self, model: type[ModelType]):
        self.model = model

    def get(self, db: Session, id_val: Any) -> ModelType | None:
        return db.get(self.model, id_val)

    def get_multi(
        self, db: Session, skip: int = 0, limit: int = 100
    ) -> list[ModelType]:
        stmt = select(self.model).offset(skip).limit(limit)
        return list(db.scalars(stmt).all())

    def create(self, db: Session, obj_in: CreateSchemaType) -> ModelType:
        obj_data = obj_in.model_dump()
        db_obj = self.model(**obj_data)
        db.add(db_obj)
        db.commit()
        db.refresh(db_obj)
        return db_obj

    def update(
        self, db: Session, db_obj: ModelType, obj_in: UpdateSchemaType | dict[str, Any]
    ) -> ModelType:
        obj_data = (
            obj_in
            if isinstance(obj_in, dict)
            else obj_in.model_dump(exclude_unset=True)
        )
        for field, value in obj_data.items():
            setattr(db_obj, field, value)
        db.commit()
        db.refresh(db_obj)
        return db_obj

    def remove(self, db: Session, id_val: Any) -> ModelType | None:
        obj = db.get(self.model, id_val)
        if obj:
            db.delete(obj)
            db.commit()
        return obj

from fastapi import HTTPException, status
from sqlalchemy.exc import IntegrityError

def _conflicto_integridad(db: Session, mensaje: str):
    """Revierte la transacción y devuelve un 409 explicando que hay registros relacionados."""
    db.rollback()
    raise HTTPException(status_code=status.HTTP_409_CONFLICT, detail=mensaje)


class AcademicoService:
    # --- NIVELES ---
    def get_niveles(self, db: Session):
        return nivel_repository.get_multi(db)

    def create_nivel(self, db: Session, nivel_in: NivelEducativoCreate):
        return nivel_repository.create(db, nivel_in)

    def update_nivel(self, db: Session, id_nivel: int, nivel_in: NivelEducativoUpdate):
        nivel = nivel_repository.get(db, id_nivel)
        if not nivel:
            raise HTTPException(
                status_code=404,
                detail=f"El nivel educativo con ID {id_nivel} no existe",
            )
        return nivel_repository.update(db, nivel, nivel_in)

    def delete_nivel(self, db: Session, id_nivel: int):
        nivel = nivel_repository.get(db, id_nivel)
        if not nivel:
            raise HTTPException(
                status_code=404,
                detail=f"El nivel educativo con ID {id_nivel} no existe",
            )
        try:
            nivel_repository.remove(db, id_nivel)
        except IntegrityError:
            _conflicto_integridad(
                db,
                "No se puede eliminar el nivel porque tiene cursos o grupos asociados",
            )
        return nivel

    # --- CURSOS ---
    def get_cursos(self, db: Session, id_nivel: int = None):
        if id_nivel:
            return curso_repository.get_by_nivel(db, id_nivel)
        return curso_repository.get_multi(db)

    def create_curso(self, db: Session, curso_in: CursoCreate):
        if not nivel_repository.get(db, curso_in.id_nivel):
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El nivel educativo con ID {curso_in.id_nivel} no existe",
            )
        return curso_repository.create(db, curso_in)

    def update_curso(self, db: Session, id_curso: int, curso_in: CursoUpdate):
        curso = curso_repository.get(db, id_curso)
        if not curso:
            raise HTTPException(
                status_code=404, detail=f"El curso con ID {id_curso} no existe"
            )
        datos = curso_in.model_dump(exclude_unset=True)
        nuevo_nivel = datos.get("id_nivel")
        if nuevo_nivel is not None and not nivel_repository.get(db, nuevo_nivel):
            raise HTTPException(
                status_code=404,
                detail=f"El nivel educativo con ID {nuevo_nivel} no existe",
            )
        return curso_repository.update(db, curso, curso_in)

    def delete_curso(self, db: Session, id_curso: int):
        curso = curso_repository.get(db, id_curso)
        if not curso:
            raise HTTPException(
                status_code=404, detail=f"El curso con ID {id_curso} no existe"
            )
        try:
            curso_repository.remove(db, id_curso)
        except IntegrityError:
            _conflicto_integridad(
                db,
                "No se puede eliminar el curso porque tiene alumnos o materias asignadas",
            )
        return curso

    # --- MATERIAS ---
    def get_materias(self, db: Session):
        return materia_repository.get_multi(db)

    def create_materia(self, db: Session, materia_in: MateriaCreate):
        return materia_repository.create(db, materia_in)

    def update_materia(self, db: Session, id_materia: int, materia_in: MateriaUpdate):
        materia = materia_repository.get(db, id_materia)
        if not materia:
            raise HTTPException(
                status_code=404, detail=f"La materia con ID {id_materia} no existe"
            )
        return materia_repository.update(db, materia, materia_in)

    def delete_materia(self, db: Session, id_materia: int):
        materia = materia_repository.get(db, id_materia)
        if not materia:
            raise HTTPException(
                status_code=404, detail=f"La materia con ID {id_materia} no existe"
            )
        try:
            materia_repository.remove(db, id_materia)
        except IntegrityError:
            _conflicto_integridad(
                db,
                "No se puede eliminar la materia porque está asignada a cursos o tiene calificaciones",
            )
        return materia

    # --- MATERIA - CURSO - PROFESOR ---
    def asignar_materia_profesor(
        self, db: Session, asignacion_in: MateriaCursoProfesorCreate
    ):
        if not materia_repository.get(db, asignacion_in.id_materia):
            raise HTTPException(
                status_code=404, detail="La materia especificada no existe"
            )
        if not curso_repository.get(db, asignacion_in.id_curso):
            raise HTTPException(
                status_code=404, detail="El curso especificado no existe"
            )
        if not profesor_repository.get_by_legajo(db, asignacion_in.legajo_profesor):
            raise HTTPException(
                status_code=404, detail="El profesor especificado no existe"
            )

        existente = materia_curso_profesor_repository.get_by_materia_y_curso(
            db, asignacion_in.id_materia, asignacion_in.id_curso
        )
        if existente:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Esta materia ya se encuentra asignada en este curso",
            )
        return materia_curso_profesor_repository.create(db, asignacion_in)

    def update_asignacion(
        self,
        db: Session,
        id_materia: int,
        id_curso: int,
        asignacion_in: MateriaCursoProfesorUpdate,
    ):
        existente = materia_curso_profesor_repository.get_by_materia_y_curso(
            db, id_materia, id_curso
        )
        if not existente:
            raise HTTPException(
                status_code=404,
                detail=f"No existe la asignación materia #{id_materia} / curso #{id_curso}",
            )
        datos = asignacion_in.model_dump(exclude_unset=True)
        nuevo_id_materia = datos.get("id_materia", id_materia)
        nuevo_id_curso = datos.get("id_curso", id_curso)
        nuevo_legajo = datos.get("legajo_profesor", existente.legajo_profesor)

        if not materia_repository.get(db, nuevo_id_materia):
            raise HTTPException(
                status_code=404, detail="La materia especificada no existe"
            )
        if not curso_repository.get(db, nuevo_id_curso):
            raise HTTPException(
                status_code=404, detail="El curso especificado no existe"
            )
        if not profesor_repository.get_by_legajo(db, nuevo_legajo):
            raise HTTPException(
                status_code=404, detail="El profesor especificado no existe"
            )

        if (nuevo_id_materia, nuevo_id_curso) != (id_materia, id_curso):
            # La clave primaria no se puede modificar en sitio: se elimina y se recrea
            duplicado = materia_curso_profesor_repository.get_by_materia_y_curso(
                db, nuevo_id_materia, nuevo_id_curso
            )
            if duplicado:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="La materia y el curso destino ya tienen una asignación existente",
                )
            materia_curso_profesor_repository.remove(db, (id_materia, id_curso))
            return materia_curso_profesor_repository.create(
                db,
                MateriaCursoProfesorCreate(
                    id_materia=nuevo_id_materia,
                    id_curso=nuevo_id_curso,
                    legajo_profesor=nuevo_legajo,
                ),
            )

        return materia_curso_profesor_repository.update(
            db, existente, {"legajo_profesor": nuevo_legajo}
        )

import os
import smtplib
from email.message import EmailMessage

# ⚠️ NOTA: Usar variables de entorno en producción. Acá podés hardcodear tu correo para probar.
SMTP_SERVER = "smtp.gmail.com"
SMTP_PORT = 587
EMAIL_SENDER = os.getenv("EMAIL_SENDER", "tu_correo@gmail.com") 
EMAIL_PASSWORD = os.getenv("EMAIL_PASSWORD", "tu_clave_de_aplicacion_gmail") 

def enviar_correo_con_pdf(destinatario: str, asunto: str, cuerpo: str, ruta_pdf: str = None):
    """Envía un correo electrónico con un archivo adjunto PDF (y luego borra el PDF temporal)."""
    if not destinatario:
        return

    msg = EmailMessage()
    msg['Subject'] = asunto
    msg['From'] = EMAIL_SENDER
    msg['To'] = destinatario
    msg.set_content(cuerpo)

    # Adjuntar PDF si existe
    if ruta_pdf and os.path.exists(ruta_pdf):
        with open(ruta_pdf, 'rb') as f:
            pdf_data = f.read()
            pdf_name = os.path.basename(ruta_pdf)
        
        msg.add_attachment(pdf_data, maintype='application', subtype='pdf', filename=pdf_name)

    try:
        with smtplib.SMTP(SMTP_SERVER, SMTP_PORT) as server:
            server.starttls()
            server.login(EMAIL_SENDER, EMAIL_PASSWORD)
            server.send_message(msg)
        print(f"📧 Correo enviado exitosamente a {destinatario}")
    except Exception as e:
        print(f"❌ Error enviando correo a {destinatario}: {e}")
    finally:
        # Limpiar el PDF temporal de tu disco duro
        if ruta_pdf and os.path.exists(ruta_pdf):
            os.remove(ruta_pdf)

from datetime import date

from fastapi import HTTPException
from sqlalchemy.orm import Session

from backend.app.models.finanzas import ConfiguracionFinanciera, Factura, ItemFactura
from backend.app.repositories import (
    alumno_repository,
    deporte_repository,
    grupo_deporte_repository,
    inscripcion_comedor_repository,
    inscripcion_deporte_repository,
    inscripcion_transporte_repository,
    recorrido_transporte_repository,
)


class FinanzasService:
    def get_valor_configuracion(self, db: Session, concepto: str) -> float:
        """Obtiene un valor fijo de la configuración (Ej: CUOTA_BASE)."""
        config = db.query(ConfiguracionFinanciera).filter_by(concepto=concepto).first()
        return float(config.valor) if config else 0.0

    def generar_facturas_mes(self, db: Session, mes: int, anio: int):
        """Genera las facturas detalladas para todos los alumnos activos."""

        # 1. Obtener valores fijos (Cuota y Comedor)
        precio_cuota_base = self.get_valor_configuracion(db, "CUOTA_BASE")
        precio_comedor = self.get_valor_configuracion(db, "COMEDOR")

        if precio_cuota_base == 0:
            raise HTTPException(
                status_code=400,
                detail="Debe configurar el valor de CUOTA_BASE en el sistema",
            )

        alumnos = alumno_repository.get_activos(db, limit=5000)
        facturas_generadas = 0

        # El vencimiento de la factura será el día 10 del mes siguiente
        mes_vencimiento = 1 if mes == 12 else mes + 1
        anio_vencimiento = anio + 1 if mes == 12 else anio
        vencimiento = date(anio_vencimiento, mes_vencimiento, 10)

        for alumno in alumnos:
            # Evitar duplicar la factura si ya se generó este mes
            existe = (
                db.query(Factura)
                .filter_by(legajo_alumno=alumno.legajo, mes=mes, anio=anio)
                .first()
            )
            if existe:
                continue

            items = []
            total = 0.0

            # A. Agregar Cuota Base (Obligatorio)
            items.append(
                ItemFactura(concepto="Cuota Base Mensual", monto=precio_cuota_base)
            )
            total += precio_cuota_base

            # B. Agregar Comedor (Si está activo)
            insc_comedor = inscripcion_comedor_repository.get_by_alumno(
                db, alumno.legajo
            )
            if insc_comedor and insc_comedor.estado == "ACTIVO" and precio_comedor > 0:
                items.append(
                    ItemFactura(concepto="Servicio de Comedor", monto=precio_comedor)
                )
                total += precio_comedor

            # C. Agregar Transporte (Si está activo)
            insc_transporte = inscripcion_transporte_repository.get_by_alumno(
                db, alumno.legajo
            )
            if insc_transporte and insc_transporte.estado == "ACTIVO":
                recorrido = recorrido_transporte_repository.get(
                    db, insc_transporte.id_recorrido
                )
                if recorrido and recorrido.precio_mensual > 0:
                    items.append(
                        ItemFactura(
                            concepto=f"Transporte: {recorrido.descripcion}",
                            monto=recorrido.precio_mensual,
                        )
                    )
                    total += float(recorrido.precio_mensual)

            # D. Agregar Deportes (Máximo 2, según regla de negocio)
            insc_deportes = inscripcion_deporte_repository.get_by_alumno(
                db, alumno.legajo
            )
            for insc_dep in insc_deportes:
                grupo = grupo_deporte_repository.get(db, insc_dep.id_grupo)
                if grupo:
                    deporte = deporte_repository.get(db, grupo.id_deporte)
                    if deporte and deporte.precio_mensual > 0:
                        items.append(
                            ItemFactura(
                                concepto=f"Deporte: {deporte.nombre}",
                                monto=deporte.precio_mensual,
                            )
                        )
                        total += float(deporte.precio_mensual)

            # 3. Crear la factura en la BD
            factura = Factura(
                legajo_alumno=alumno.legajo,
                mes=mes,
                anio=anio,
                total=total,
                estado="PENDIENTE",
                fecha_emision=date.today(),
                fecha_vencimiento=vencimiento,
            )
            db.add(factura)
            db.flush()  # Guarda temporalmente para que PostgreSQL asigne el id_factura

            # 4. Vincular el detalle (los ítems) a la factura creada
            for item in items:
                item.id_factura = factura.id_factura
                db.add(item)

            facturas_generadas += 1

        # Confirmar todos los cambios en la base de datos de una sola vez
        db.commit()
        return {
            "mensaje": f"Proceso finalizado. Se generaron {facturas_generadas} facturas nuevas."
        }

import os
from datetime import datetime, timedelta, timezone

import bcrypt
import jwt
from fastapi import HTTPException, status
from sqlalchemy.orm import Session

from backend.app.models.personas import Padre, PadreAlumno
from backend.app.models.seguridad import Usuario
from backend.app.repositories import (
    alumno_repository,
    padre_alumno_repository,
    padre_repository,
    profesor_repository,
    rol_repository,
    usuario_repository,

from backend.app.schemas.personas import PadreCreate
from backend.app.schemas.seguridad import (
    Token,
    UsuarioAdminResponse,
    UsuarioCreate,
    UsuarioUpdate,
)

SECRET_KEY = os.getenv("SECRET_KEY")
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = int(
    os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "1440")
)  # 24 horas


class SeguridadService:
    @staticmethod
    def verify_password(plain_password: str, hashed_password: str) -> bool:
        """Verifica una contraseña plana contra el hash bcrypt almacenado."""
        if not plain_password or not hashed_password:
            return False
        try:
            password_bytes = plain_password.encode("utf-8")[:72]
            hashed_bytes = (
                hashed_password.encode("utf-8")
                if isinstance(hashed_password, str)
                else hashed_password
            )
            return bcrypt.checkpw(password_bytes, hashed_bytes)
        except Exception:
            return False

    @staticmethod
    def get_password_hash(password: str) -> str:
        """Genera un hash bcrypt seguro en formato string ($2b$)."""
        password_bytes = password.encode("utf-8")[:72]
        salt = bcrypt.gensalt()
        return bcrypt.hashpw(password_bytes, salt).decode("utf-8")

    @staticmethod
    def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
        to_encode = data.copy()
        expire = datetime.now(timezone.utc) + (
            expires_delta or timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
        )
        to_encode.update({"exp": expire})
        return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

    def authenticate_user(self, db: Session, username: str, password: str):
        user = usuario_repository.get_by_username(db, username=username)
        if not user or not self.verify_password(password, user.password_hash):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Credenciales incorrectas",
                headers={"WWW-Authenticate": "Bearer"},
            )
        if not user.activo:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="El usuario se encuentra inactivo",
            )
        return user

    def login(self, db: Session, username: str, password: str) -> Token:
        user = self.authenticate_user(db, username, password)
        rol = rol_repository.get(db, user.id_rol)
        rol_nombre = rol.nombre if rol else "USER"

        token_data = {
            "sub": user.username,
            "id_usuario": user.id_usuario,
            "rol": rol_nombre,
        }
        access_token = self.create_access_token(data=token_data)
        return Token(
            access_token=access_token,
            token_type="bearer",
            rol=rol_nombre,
            username=user.username,
        )

    def get_usuario_from_token(self, db: Session, token: str) -> Usuario:
        """Decodifica el JWT y devuelve la entidad Usuario. Lanza 401/403 si no es válido."""
        try:
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        except jwt.ExpiredSignatureError:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="El token expiró. Inicie sesión nuevamente.",
                headers={"WWW-Authenticate": "Bearer"},
            )
        except jwt.InvalidTokenError:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Token inválido",
                headers={"WWW-Authenticate": "Bearer"},
            )

        username = payload.get("sub")
        user = usuario_repository.get_by_username(db, username) if username else None
        if not user:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="El usuario del token no existe",
                headers={"WWW-Authenticate": "Bearer"},
            )
        if not user.activo:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="El usuario se encuentra inactivo",
            )
        return user

    def get_rol_nombre(self, db: Session, user: Usuario) -> str:
        """Devuelve el nombre del rol (ADMIN, TEACHER, PARENT, STUDENT) del usuario."""
        rol = rol_repository.get(db, user.id_rol)
        return rol.nombre if rol else ""

    def get_perfil_from_token(self, db: Session, token: str) -> UsuarioAdminResponse:
        """Decodifica el JWT y devuelve el perfil del usuario autenticado."""
        user = self.get_usuario_from_token(db, token)
        return self._to_admin_response(db, user)

    def change_password(
        self, db: Session, user: Usuario, password_actual: str, password_nueva: str
    ) -> dict:
        """Cambia la contraseña del propio usuario autenticado."""
        if not self.verify_password(password_actual, user.password_hash):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="La contraseña actual no es correcta",
            )
        if len(password_nueva or "") < 6:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="La nueva contraseña debe tener al menos 6 caracteres",
            )
        user.password_hash = self.get_password_hash(password_nueva)
        db.commit()
        return {"detail": "Contraseña actualizada correctamente"}

    def _validar_alumnos_vinculables(
        self, db: Session, legajos: list[int]
    ) -> list[int]:
        """Valida que todos los legajos existan y devuelve una lista sin repetidos."""
        resultado: list[int] = []
        for legajo in legajos:
            legajo_int = int(legajo)
            if legajo_int in resultado:
                continue
            if not alumno_repository.get_by_legajo(db, legajo_int):
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail=f"El alumno con legajo #{legajo_int} no existe",
                )
            resultado.append(legajo_int)
        return resultado

    def _sincronizar_alumnos_tutor(
        self, db: Session, id_padre: int | None, legajos: list[int]
    ):
        """Reemplaza los alumnos vinculados al tutor indicado.

        No hace commit: se confirma junto con el resto de la operación del usuario.
        """
        legajos_validos = self._validar_alumnos_vinculables(db, legajos or [])
        if legajos_validos and not id_padre:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Debe indicar los datos del tutor antes de vincular alumnos",
            )
        if not id_padre:
            return
        if not padre_repository.get(db, id_padre):
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El tutor con ID {id_padre} no existe",
            )

        relaciones = padre_alumno_repository.get_alumnos_por_padre(db, id_padre)
        actuales = {relacion.legajo_alumno for relacion in relaciones}
        deseados = set(legajos_validos)

        for relacion in relaciones:
            if relacion.legajo_alumno not in deseados:
                db.delete(relacion)
        for legajo in deseados - actuales:
            db.add(PadreAlumno(id_padre=id_padre, legajo_alumno=legajo))

    def _obtener_o_crear_tutor(self, db: Session, tutor_in: PadreCreate) -> Padre:
        """Reutiliza el tutor con ese DNI o lo crea en la misma transacción."""
        padre = padre_repository.get_by_dni(db, tutor_in.dni)
        if padre:
            for campo, valor in tutor_in.model_dump().items():
                if valor is not None:
                    setattr(padre, campo, valor)
            return padre
        padre = Padre(**tutor_in.model_dump())
        db.add(padre)
        db.flush()
        return padre

    def _actualizar_tutor(
        self, db: Session, padre: Padre, tutor_in: PadreCreate
    ) -> None:
        """Actualiza los datos del tutor vinculado al usuario."""
        existente = padre_repository.get_by_dni(db, tutor_in.dni)
        if existente and existente.id_padre != padre.id_padre:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Ya existe otro tutor con el DNI '{tutor_in.dni}'",
            )
        for campo, valor in tutor_in.model_dump().items():
            setattr(padre, campo, valor)

    def registrar_usuario(self, db: Session, user_in: UsuarioCreate):
        if usuario_repository.get_by_username(db, user_in.username):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"El nombre de usuario '{user_in.username}' ya está registrado",
            )
        rol = rol_repository.get(db, user_in.id_rol)
        if not rol:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El rol con ID {user_in.id_rol} no existe",
            )

        if user_in.tutor is not None:
            if rol.nombre != "PARENT":
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Solo los usuarios con rol Padre / Tutor pueden tener datos de tutor",
                )
            if user_in.id_padre is not None:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=(
                        "Indique un tutor existente o los datos de un tutor nuevo, "
                        "no ambos"
                    ),
                )

        legajos_alumnos = user_in.legajos_alumnos
        if legajos_alumnos:
            if rol.nombre != "PARENT":
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Solo los usuarios con rol Padre / Tutor pueden tener alumnos vinculados",
                )
            if not user_in.id_padre and user_in.tutor is None:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Debe indicar los datos del tutor para vincular alumnos",
                )
            self._validar_alumnos_vinculables(db, legajos_alumnos)

        if user_in.tutor is not None:
            padre = self._obtener_o_crear_tutor(db, user_in.tutor)
            user_in = user_in.model_copy(update={"id_padre": padre.id_padre})

        hashed_pwd = self.get_password_hash(user_in.password)
        user = usuario_repository.create_user_with_hash(db, user_in, hashed_pwd)
        if legajos_alumnos:
            self._sincronizar_alumnos_tutor(db, user.id_padre, legajos_alumnos)
        db.commit()
        db.refresh(user)
        return user

    # ================= ADMINISTRACIÓN DE USUARIOS =================
    def get_roles(self, db: Session):
        return rol_repository.get_multi(db)

    def _to_admin_response(self, db: Session, user) -> UsuarioAdminResponse:
        rol = rol_repository.get(db, user.id_rol)
        rol_nombre = rol.nombre if rol else ""

        alumnos_vinculados: list[int] = []
        if user.id_padre:
            alumnos_vinculados = [
                relacion.legajo_alumno
                for relacion in padre_alumno_repository.get_alumnos_por_padre(
                    db, user.id_padre
                )
            ]

        if user.legajo_alumno:
            alumno = alumno_repository.get_by_legajo(db, user.legajo_alumno)
            vinculacion = (
                f"Alumno: {alumno.apellido}, {alumno.nombre} (#{alumno.legajo})"
                if alumno
                else f"Alumno #{user.legajo_alumno}"
            )
        elif user.legajo_profesor:
            profesor = profesor_repository.get_by_legajo(db, user.legajo_profesor)
            vinculacion = (
                f"Docente: {profesor.apellido}, {profesor.nombre} (#{profesor.legajo})"
                if profesor
                else f"Docente #{user.legajo_profesor}"
            )
        elif user.id_padre:
            padre = padre_repository.get(db, user.id_padre)
            vinculacion = (
                f"Tutor: {padre.apellido}, {padre.nombre}"
                if padre
                else f"Tutor #{user.id_padre}"
            )
            cantidad = len(alumnos_vinculados)
            vinculacion += (
                f" · {cantidad} alumno{'s' if cantidad != 1 else ''} vinculado"
                f"{'s' if cantidad != 1 else ''}"
                if cantidad
                else " · Sin alumnos vinculados"
            )
        elif rol_nombre == "ADMIN":
            vinculacion = "Acceso Total"
        else:
            vinculacion = "Sin vínculo"

        return UsuarioAdminResponse(
            id_usuario=user.id_usuario,
            username=user.username,
            id_rol=user.id_rol,
            legajo_alumno=user.legajo_alumno,
            legajo_profesor=user.legajo_profesor,
            id_padre=user.id_padre,
            activo=user.activo,
            creado_en=user.creado_en,
            rol=rol_nombre,
            vinculacion=vinculacion,
            alumnos_vinculados=alumnos_vinculados,
        )

    def get_usuarios(self, db: Session) -> list[UsuarioAdminResponse]:
        usuarios = usuario_repository.get_multi(db, limit=200)
        return [self._to_admin_response(db, u) for u in usuarios]

    def update_usuario(
        self, db: Session, id_usuario: int, user_in: UsuarioUpdate
    ) -> UsuarioAdminResponse:
        user = usuario_repository.get(db, id_usuario)
        if not user:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El usuario con ID {id_usuario} no existe",
            )

        data = user_in.model_dump(exclude_unset=True)
        password = data.pop("password", None)
        legajos_alumnos = data.pop("legajos_alumnos", None)
        tutor_data = data.pop("tutor", None)
        if tutor_data is not None:
            tutor_data = PadreCreate(**tutor_data)
        if password:
            user.password_hash = self.get_password_hash(password)

        nuevo_username = data.get("username")
        if nuevo_username and nuevo_username != user.username:
            if usuario_repository.get_by_username(db, nuevo_username):
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=f"El nombre de usuario '{nuevo_username}' ya está registrado",
                )

        nuevo_rol = data.get("id_rol")
        rol_efectivo = (
            rol_repository.get(db, nuevo_rol)
            if nuevo_rol is not None
            else rol_repository.get(db, user.id_rol)
        )
        if nuevo_rol is not None and not rol_efectivo:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"El rol con ID {nuevo_rol} no existe",
            )

        campos_vinculo = {"legajo_alumno", "legajo_profesor", "id_padre"}

        if tutor_data is not None:
            if not rol_efectivo or rol_efectivo.nombre != "PARENT":
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=(
                        "Solo los usuarios con rol Padre / Tutor pueden tener "
                        "datos de tutor"
                    ),
                )
            if campos_vinculo & data.keys():
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=(
                        "No combine los datos del tutor con otros vínculos "
                        "de alumno o docente"
                    ),
                )

            # El usuario queda vinculado al tutor, nunca a un alumno o docente.
            user.legajo_alumno = None
            user.legajo_profesor = None
            if user.id_padre:
                padre = padre_repository.get(db, user.id_padre)
                if not padre:
                    raise HTTPException(
                        status_code=status.HTTP_404_NOT_FOUND,
                        detail=f"El tutor con ID {user.id_padre} no existe",
                    )
                self._actualizar_tutor(db, padre, tutor_data)
            else:
                padre = self._obtener_o_crear_tutor(db, tutor_data)
                user.id_padre = padre.id_padre

        # Los vínculos (alumno/docente/tutor) se gestionan en bloque: si se envía
        # alguno, se limpian los demás para que una cuenta tenga un único vínculo.
        elif campos_vinculo & data.keys():
            vinculos = {campo: data.pop(campo, None) for campo in list(campos_vinculo)}

            if vinculos.get("legajo_alumno") is not None and not (
                alumno_repository.get_by_legajo(db, vinculos["legajo_alumno"])
            ):
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail=(
                        f"El alumno con legajo #{vinculos['legajo_alumno']} no existe"
                    ),
                )
            if vinculos.get("legajo_profesor") is not None and not (
                profesor_repository.get_by_legajo(db, vinculos["legajo_profesor"])
            ):
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail=(
                        "El docente con legajo "
                        f"#{vinculos['legajo_profesor']} no existe"
                    ),
                )
            if vinculos.get("id_padre") is not None and not (
                padre_repository.get(db, vinculos["id_padre"])
            ):
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail=f"El tutor con ID {vinculos['id_padre']} no existe",
                )

            for campo in campos_vinculo:
                setattr(user, campo, vinculos.get(campo))

        for field, value in data.items():
            if value is not None:
                setattr(user, field, value)

        if legajos_alumnos is not None:
            if rol_efectivo and rol_efectivo.nombre == "PARENT":
                self._sincronizar_alumnos_tutor(db, user.id_padre, legajos_alumnos)
            elif legajos_alumnos:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=(
                        "Solo los usuarios con rol Padre / Tutor pueden tener "
                        "alumnos vinculados"
                    ),
                )

        db.commit()
        db.refresh(user)
        return self._to_admin_response(db, user)

class ServiciosEscolaresService:
    # --- DEPORTES Y GRUPOS ---
    def create_deporte(self, db: Session, deporte_in: DeporteCreate):
        return deporte_repository.create(db, deporte_in)

    def get_deportes(self, db: Session):
        return deporte_repository.get_multi(db)

    def update_deporte(self, db: Session, id_deporte: int, deporte_in: DeporteUpdate):
        deporte = deporte_repository.get(db, id_deporte)
        if not deporte:
            raise HTTPException(
                status_code=404, detail=f"El deporte con ID {id_deporte} no existe"
            )
        return deporte_repository.update(db, deporte, deporte_in)

    def delete_deporte(self, db: Session, id_deporte: int):
        deporte = deporte_repository.get(db, id_deporte)
        if not deporte:
            raise HTTPException(
                status_code=404, detail=f"El deporte con ID {id_deporte} no existe"
            )
        try:
            deporte_repository.remove(db, id_deporte)
        except IntegrityError:
            db.rollback()
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="No se puede eliminar el deporte porque tiene grupos o inscriptos asociados",
            )
        return deporte

    def create_grupo_deporte(self, db: Session, grupo_in: GrupoDeporteCreate):
        if not deporte_repository.get(db, grupo_in.id_deporte):
            raise HTTPException(status_code=404, detail="Deporte no encontrado")
        if not nivel_repository.get(db, grupo_in.id_nivel):
            raise HTTPException(status_code=404, detail="Nivel educativo no encontrado")
        if not horario_repository.get(db, grupo_in.id_horario):
            raise HTTPException(status_code=404, detail="Horario no encontrado")
        return grupo_deporte_repository.create(db, grupo_in)

    def update_grupo_deporte(
        self, db: Session, id_grupo: int, grupo_in: GrupoDeporteUpdate
    ):
        grupo = grupo_deporte_repository.get(db, id_grupo)
        if not grupo:
            raise HTTPException(
                status_code=404, detail=f"El grupo con ID {id_grupo} no existe"
            )
        datos = grupo_in.model_dump(exclude_unset=True)
        if datos.get("id_deporte") is not None and not deporte_repository.get(
            db, datos["id_deporte"]
        ):
            raise HTTPException(status_code=404, detail="Deporte no encontrado")
        if datos.get("id_nivel") is not None and not nivel_repository.get(
            db, datos["id_nivel"]
        ):
            raise HTTPException(status_code=404, detail="Nivel educativo no encontrado")
        if datos.get("id_horario") is not None and not horario_repository.get(
            db, datos["id_horario"]
        ):
            raise HTTPException(status_code=404, detail="Horario no encontrado")
        return grupo_deporte_repository.update(db, grupo, grupo_in)

    def delete_grupo_deporte(self, db: Session, id_grupo: int):
        grupo = grupo_deporte_repository.get(db, id_grupo)
        if not grupo:
            raise HTTPException(
                status_code=404, detail=f"El grupo con ID {id_grupo} no existe"
            )
        try:
            grupo_deporte_repository.remove(db, id_grupo)
        except IntegrityError:
            db.rollback()
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="No se puede eliminar el grupo porque tiene alumnos inscriptos",
            )
        return grupo

    def inscribir_deporte(self, db: Session, inscripcion_in: InscripcionDeporteCreate):
        if not alumno_repository.get_by_legajo(db, inscripcion_in.legajo_alumno):
            raise HTTPException(status_code=404, detail="El alumno no existe")

        nuevo_grupo = grupo_deporte_repository.get(db, inscripcion_in.id_grupo)
        if not nuevo_grupo:
            raise HTTPException(status_code=404, detail="El grupo deportivo no existe")

        if inscripcion_deporte_repository.get_inscripcion(
            db, inscripcion_in.legajo_alumno, inscripcion_in.id_grupo
        ):
            raise HTTPException(
                status_code=400, detail="El alumno ya está inscripto en este grupo"
            )

        # 1. Validar límite máximo de 2 deportes (RF03)
        inscripciones_actuales = inscripcion_deporte_repository.get_by_alumno(
            db, inscripcion_in.legajo_alumno
        )
        if len(inscripciones_actuales) >= 2:
            raise HTTPException(
                status_code=400,
                detail="El alumno ya alcanzó el límite máximo de dos deportes simultáneos",
            )

        # 2. Controlar conflictos de horario (RF07)
        for inscripcion in inscripciones_actuales:
            grupo_actual = grupo_deporte_repository.get(db, inscripcion.id_grupo)
            if grupo_actual and grupo_actual.id_horario == nuevo_grupo.id_horario:
                raise HTTPException(
                    status_code=409,
                    detail="Existe un conflicto de horarios con otro deporte en el que el alumno ya está inscripto",
                )

        return inscripcion_deporte_repository.create(db, inscripcion_in)

    # --- TRANSPORTE ---
    def create_recorrido(self, db: Session, recorrido_in: RecorridoTransporteCreate):
        return recorrido_transporte_repository.create(db, recorrido_in)

    def get_recorridos(self, db: Session):
        return recorrido_transporte_repository.get_multi(db)

    def update_recorrido(
        self, db: Session, id_recorrido: int, recorrido_in: RecorridoTransporteUpdate
    ):
        recorrido = recorrido_transporte_repository.get(db, id_recorrido)
        if not recorrido:
            raise HTTPException(
                status_code=404, detail=f"El recorrido con ID {id_recorrido} no existe"
            )
        return recorrido_transporte_repository.update(db, recorrido, recorrido_in)

    def delete_recorrido(self, db: Session, id_recorrido: int):
        recorrido = recorrido_transporte_repository.get(db, id_recorrido)
        if not recorrido:
            raise HTTPException(
                status_code=404, detail=f"El recorrido con ID {id_recorrido} no existe"
            )
        try:
            recorrido_transporte_repository.remove(db, id_recorrido)
        except IntegrityError:
            db.rollback()
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail="No se puede eliminar el recorrido porque tiene pasajeros asociados",
            )
        return recorrido

    def inscribir_transporte(
        self, db: Session, inscripcion_in: InscripcionTransporteCreate
    ):
        if not alumno_repository.get_by_legajo(db, inscripcion_in.legajo_alumno):
            raise HTTPException(status_code=404, detail="El alumno no existe")
        if not recorrido_transporte_repository.get(db, inscripcion_in.id_recorrido):
            raise HTTPException(status_code=404, detail="El recorrido no existe")

        existente = inscripcion_transporte_repository.get_by_alumno(
            db, inscripcion_in.legajo_alumno
        )
        if existente and existente.estado == "ACTIVO":
            raise HTTPException(
                status_code=400,
                detail="El alumno ya cuenta con inscripción activa en transporte",
            )
        return inscripcion_transporte_repository.create(db, inscripcion_in)

    def baja_transporte(self, db: Session, legajo_alumno: int):
        res = inscripcion_transporte_repository.soft_delete(db, legajo_alumno)
        if not res:
            raise HTTPException(
                status_code=404, detail="Inscripción de transporte no encontrada"
            )
        return res

    # --- COMEDOR ---
    def inscribir_comedor(self, db: Session, inscripcion_in: InscripcionComedorCreate):
        if not alumno_repository.get_by_legajo(db, inscripcion_in.legajo_alumno):
            raise HTTPException(status_code=404, detail="El alumno no existe")

        existente = inscripcion_comedor_repository.get_by_alumno(
            db, inscripcion_in.legajo_alumno
        )
        if existente and existente.estado == "ACTIVO":
            raise HTTPException(
                status_code=400,
                detail="El alumno ya cuenta con inscripción activa en comedor",
            )
        return inscripcion_comedor_repository.create(db, inscripcion_in)

    def baja_comedor(self, db: Session, legajo_alumno: int):
        res = inscripcion_comedor_repository.soft_delete(db, legajo_alumno)
        if not res:
            raise HTTPException(
                status_code=404, detail="Inscripción de comedor no encontrada"
            )
        return res

    # ================= DETALLES PARA EL PANEL ADMIN =================
    @staticmethod
    def _nombre_alumno(db: Session, legajo_alumno: int) -> str:
        alumno = alumno_repository.get_by_legajo(db, legajo_alumno)
        if not alumno:
            return f"Legajo #{legajo_alumno}"
        return f"{alumno.apellido}, {alumno.nombre} (#{alumno.legajo})"

    def get_grupos_deporte(self, db: Session) -> list[GrupoDeporteDetalleResponse]:
        grupos: list[GrupoDeporteDetalleResponse] = []
        for grupo in grupo_deporte_repository.get_multi(db, limit=200):
            deporte = deporte_repository.get(db, grupo.id_deporte)
            nivel = nivel_repository.get(db, grupo.id_nivel)
            horario = horario_repository.get(db, grupo.id_horario)
            grupos.append(
                GrupoDeporteDetalleResponse(
                    id_grupo=grupo.id_grupo,
                    id_deporte=grupo.id_deporte,
                    deporte=deporte.nombre
                    if deporte
                    else f"Deporte #{grupo.id_deporte}",
                    id_nivel=grupo.id_nivel,
                    nivel=nivel.nombre if nivel else f"Nivel #{grupo.id_nivel}",
                    id_horario=grupo.id_horario,
                    horario=horario.descripcion
                    if horario
                    else f"Horario #{grupo.id_horario}",
                )
            )
        return grupos

    def get_inscripciones_deporte(
        self, db: Session
    ) -> list[InscripcionDeporteDetalleResponse]:
        inscripciones: list[InscripcionDeporteDetalleResponse] = []
        for inscripcion in inscripcion_deporte_repository.get_multi(db, limit=500):
            grupo = grupo_deporte_repository.get(db, inscripcion.id_grupo)
            deporte = deporte_repository.get(db, grupo.id_deporte) if grupo else None
            inscripciones.append(
                InscripcionDeporteDetalleResponse(
                    legajo_alumno=inscripcion.legajo_alumno,
                    alumno=self._nombre_alumno(db, inscripcion.legajo_alumno),
                    id_grupo=inscripcion.id_grupo,
                    deporte=deporte.nombre
                    if deporte
                    else f"Grupo #{inscripcion.id_grupo}",
                    fecha_inscripcion=inscripcion.fecha_inscripcion,
                )
            )
        return inscripciones

    def get_inscripciones_transporte(

    ) -> list[InscripcionTransporteDetalleResponse]:

                    id_recorrido=inscripcion.id_recorrido,
